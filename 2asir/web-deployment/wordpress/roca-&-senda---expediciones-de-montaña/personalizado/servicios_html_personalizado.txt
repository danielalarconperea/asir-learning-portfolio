    function filterRoutes(cat, btn) {
        // Toggle Buttons
        document.querySelectorAll('.rs-filter-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');

        // Filter Cards
        document.querySelectorAll('.rs-route-card').forEach(card => {
            if (cat === 'Todos' || card.dataset.category === cat) {
                card.style.display = 'flex';
            } else {
                card.style.display = 'none';
            }
        });
    }

    function openModal(id) {
        const data = routesData[id];
        document.getElementById('modal-img').src = data.img;
        document.getElementById('modal-title').innerText = data.title;
        document.getElementById('modal-desc').innerText = data.desc;
        document.getElementById('modal-badge-cat').innerText = data.cat;
        document.getElementById('modal-badge-diff').innerText = data.diff;
        document.getElementById('modal-duration').innerText = data.duration;
        document.getElementById('modal-price').innerText = data.price;
        
        document.getElementById('rs-modal-overlay').style.display = 'flex';
    }

    function closeModal() {
        document.getElementById('rs-modal-overlay').style.display = 'none';
    }
